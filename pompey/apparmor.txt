#include <tunables/global>

profile pompey flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>
  #include <abstractions/nameservice>

  file,
  network,
  capability net_admin,
  capability net_raw,
  capability net_bind_service,
  capability chown,
  capability fowner,
  capability fsetid,
  capability setuid,
  capability setgid,
  capability dac_override,
  capability kill,

  signal (send) set=(kill,term,int,hup,cont),

  # s6-overlay
  /init ix,
  /bin/** ix,
  /sbin/** ix,
  /usr/bin/** ix,
  /usr/sbin/** ix,
  /usr/local/bin/** ix,
  /usr/local/bin/pompey-wg/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,
  /dev/net/tun rw,

  # bashio + tmp. rwk (no chmod/exec) made GNU tar fail on /tmp with
  # "Cannot change mode ... Operation not permitted". Engine archives
  # unpack under /data/engines; keep /tmp usable for everything else.
  /usr/lib/bashio/** ix,
  /tmp/** rwlix,

  # HA options, addon config, fetched engines
  /data/** rwlix,
  /config/** rwlix,
  /share/** rw,
  /media/** rw,

  # WireGuard + NAT-PMP + kill switch (nft first; HAOS has no iptable_filter)
  /usr/bin/wg ix,
  /usr/bin/wg-quick ix,
  /usr/bin/wireguard-go ix,
  /usr/sbin/sysctl ix,
  /sbin/sysctl ix,
  /usr/bin/natpmpc ix,
  /usr/sbin/nft ix,
  /etc/wireguard/** rw,
  /etc/pompey-lan-networks r,

  # nginx
  /usr/sbin/nginx ix,
  /etc/nginx/** rw,
  /var/lib/nginx/** rw,
  /var/log/nginx/** rw,
  /run/nginx/** rwk,
  /usr/share/pompey/** r,
}
